df <- data.frame(
temp = c(79, 74, 73, 76, 82, 72, 71, 79, 83, 70, 73, 75, 77),
chirps = c(44, 40, 38, 43, 48, 37, 34, 45, 51, 32, 35, 42, 43))

r <- cor(df$temp, df$chirps) # 0.97
mean_temp <- mean(df$temp) # 75.69
sd_temp <- sd(df$temp) # 4.11
mean_chirp <- mean(df$chirps) # 40.92
sd_chirp <- sd(df$chirps) # 5.59

# To get the line of best fit, temp = 0.714 * chirps + 46.474 
# 0.714 is the slope of the line, which would come for the correlation times the rise over the run, which is the sd of the temp over the sd of the chirps, or 0.97*(4.11)/(5.59) = 0.714
slope <- r*(sd_temp/sd_chirp)
# 46.474 is the 'y intercept' of the equation, which would be the amount of temp when there is 0 chirp, which can be calculated from mean temp - slope times mean chrips
y_intercept <- mean_temp - slope * mean_chirp
